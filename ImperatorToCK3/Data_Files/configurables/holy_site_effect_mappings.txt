#
# Mapping of Imperator Religion Modifier + Deity Passive Effect -> CK3 Holy Site Modifier
#
# This is used for conversion of dynamic Holy Sites.
#
# Each link maps an Imperator effect to CK3 modifiers, spread across the holy site's
# modifier blocks. Every block can contain multiple ck3-factor pairs. The final modifier
# value is the Imperator value multiplied by the factor.
#
# link = {
#     ir = imperator_effect
#     faith_character_modifier = {
#         ck3 = modifier factor = 10
#     }
#     faith_character_modifier_scaled = {
#         scale = holy_site_total_artifact_rarity
#         ck3 = modifier factor = 0.2
#     }
#     county_holder_character_modifier = {
#         ck3 = modifier factor = 5
#     }
#     county_modifier_scaled = {
#         scale = holy_site_total_artifact_rarity
#         ck3 = modifier factor = 0.2
#     }
# }
#
# Notes:
# - faith_character_modifier(+_scaled) only take effect for eminent holy sites.
# - county_holder_character_modifier and county_modifier_scaled work for both
#   regular and eminent holy sites.
# - scale is optional for the scaled blocks; it is a scripted value or a number.
#
